@lesson async/await и Task
Асинхронная функция помечается `async`, вызывается через `await` и может бросать ошибки (`async throws`, вызов `try await`). Пока она ждёт сеть, поток свободен для других задач.
`async let` запускает несколько запросов параллельно, `TaskGroup` — динамическое число параллельных задач. Из синхронного кода асинхронную работу запускают через `Task { ... }`.
Задачи поддерживают отмену: `Task.checkCancellation()` бросит ошибку, если задачу отменили, — например, пользователь ушёл с экрана.
```code
import Foundation

func loadName() async throws -> String {
    try await Task.sleep(for: .milliseconds(200))
    return "Ада"
}

func loadScore() async throws -> Int {
    try await Task.sleep(for: .milliseconds(200))
    return 95
}

let start = Date()
Task {
    async let name = loadName()          // оба запроса идут параллельно
    async let score = loadScore()
    let (n, s) = try await (name, score)
    print(n, s, Date().timeIntervalSince(start) < 0.35)     // Ада 95 true
}
try? await Task.sleep(for: .milliseconds(500))
```
! `async let` — параллельные независимые запросы
! Ошибки из `async throws` ловят через `do/catch` с `try await`
! Проверяйте отмену в долгих задачах

@lesson URLSession и Codable
Сетевой запрос: `let (data, response) = try await URLSession.shared.data(from: url)`. Обязательно проверяйте `HTTPURLResponse.statusCode` — URLSession не считает 404 или 500 ошибкой.
`Codable` превращает JSON в структуры: структура объявляет `Decodable`, `JSONDecoder().decode(T.self, from: data)` делает разбор. `keyDecodingStrategy = .convertFromSnakeCase` переводит `created_at` в `createdAt`, даты — `dateDecodingStrategy = .iso8601`.
Для POST создают `URLRequest`: метод, заголовок `Content-Type: application/json` и тело `JSONEncoder().encode(body)`.
```code
import Foundation

struct Repo: Decodable, Identifiable {
    let id: Int
    let name: String
    let stargazersCount: Int
}

enum APIError: Error { case badStatus(Int) }

func repos(of user: String) async throws -> [Repo] {
    let url = URL(string: "https://api.github.com/users/\(user)/repos")!
    let (data, response) = try await URLSession.shared.data(from: url)
    guard let http = response as? HTTPURLResponse else { throw URLError(.badServerResponse) }
    guard (200..<300).contains(http.statusCode) else { throw APIError.badStatus(http.statusCode) }
    let decoder = JSONDecoder()
    decoder.keyDecodingStrategy = .convertFromSnakeCase
    return try decoder.decode([Repo].self, from: data)
}

let json = #"[{"id": 1, "name": "avaland", "stargazers_count": 42}]"#.data(using: .utf8)!
let dec = JSONDecoder()
dec.keyDecodingStrategy = .convertFromSnakeCase
let parsed = try dec.decode([Repo].self, from: json)
print(parsed[0].name, parsed[0].stargazersCount)        // avaland 42
```
! URLSession не бросает ошибку на 404 и 500 — проверяйте статус сами
! `convertFromSnakeCase` избавляет от ручных CodingKeys
! Ошибка декодирования — частая причина «пустого экрана», логируйте её

@lesson Слой API и обработка ошибок
Сетевой код выносят из View в сервис: протокол `CoursesAPI` с методами и реализация на URLSession. Модель экрана получает сервис через инициализатор — в тестах подставляют фальшивый.
Ошибки превращают в понятные сообщения: нет интернета (`URLError.notConnectedToInternet`), таймаут, 401 (нужно войти заново), 5xx (повторить позже). Пользователь должен видеть, что случилось, и кнопку «Повторить».
Токены авторизации добавляют в заголовок `Authorization: Bearer ...` в одном месте — в клиенте API, а сами токены хранят в Keychain, а не в UserDefaults.
```code
import Foundation

protocol CoursesAPI {
    func courses() async throws -> [String]
}

struct LiveCoursesAPI: CoursesAPI {
    let base = URL(string: "https://api.avaland.kz")!
    let token: String
    func courses() async throws -> [String] {
        var req = URLRequest(url: base.appending(path: "courses"))
        req.setValue("Bearer \(token)", forHTTPHeaderField: "Authorization")
        req.timeoutInterval = 10
        let (data, resp) = try await URLSession.shared.data(for: req)
        guard (resp as? HTTPURLResponse)?.statusCode == 200 else { throw URLError(.badServerResponse) }
        return try JSONDecoder().decode([String].self, from: data)
    }
}

func message(for error: Error) -> String {
    switch (error as? URLError)?.code {
    case .notConnectedToInternet?: return "Нет интернета"
    case .timedOut?: return "Сервер долго не отвечает"
    default: return "Что-то пошло не так. Повторите позже"
    }
}

print(message(for: URLError(.notConnectedToInternet)))    // Нет интернета
```
! Сеть — в сервисе за протоколом, не во View
! Каждой ошибке — понятный текст и кнопка «Повторить»
! Токены — в Keychain

@lesson Тесты: Swift Testing
Тесты пишут фреймворком Swift Testing: функция с `@Test`, проверки — `#expect(выражение)` и `#require` (останавливает тест). Асинхронные тесты — просто `async` функции.
Модель экрана тестируют с фальшивым API: проверяют, что после `load()` состояние стало `.loaded` с нужными данными, а при ошибке — `.failed`.
Параметризованные тесты: `@Test(arguments: [...])` прогоняет одну функцию на многих входах. Тесты запускают в Xcode (⌘U) или `swift test`.
```code
import Testing

struct FakeAPI: CoursesAPI {
    var result: Result<[String], Error>
    func courses() async throws -> [String] { try result.get() }
}

@MainActor
final class CoursesModel {
    enum State: Equatable { case loading, loaded([String]), failed }
    var state = State.loading
    let api: CoursesAPI
    init(api: CoursesAPI) { self.api = api }
    func load() async {
        do { state = .loaded(try await api.courses()) } catch { state = .failed }
    }
}

@Test @MainActor func loadsCourses() async {
    let model = CoursesModel(api: FakeAPI(result: .success(["Swift", "SwiftUI"])))
    await model.load()
    #expect(model.state == .loaded(["Swift", "SwiftUI"]))
}

@Test @MainActor func showsErrorOnFailure() async {
    struct Boom: Error {}
    let model = CoursesModel(api: FakeAPI(result: .failure(Boom())))
    await model.load()
    #expect(model.state == .failed)
}

@Test(arguments: [(2, 8, 25), (8, 8, 100), (0, 5, 0)])
func progress(done: Int, total: Int, percent: Int) {
    #expect(done * 100 / total == percent)
}
```
! Фальшивые реализации протоколов — основа тестов моделей
! `#expect` продолжает тест, `#require` останавливает
! `@Test(arguments:)` вместо копирования тестов
