@lesson database/sql и пул соединений
Пакет `database/sql` — общий интерфейс к базам, драйвер подключают отдельно (для PostgreSQL — pgx). `sql.Open` не подключается сразу, а создаёт пул соединений; проверка — `db.PingContext(ctx)`.
Пул настраивают: `SetMaxOpenConns`, `SetMaxIdleConns`, `SetConnMaxLifetime`. Объект `*sql.DB` создают один раз на приложение и передают в обработчики.
Запросы: `QueryRowContext` — одна строка, `QueryContext` — много (обязательно `defer rows.Close()` и проверка `rows.Err()`), `ExecContext` — изменения без результата.
```code
package main

import (
	"context"
	"database/sql"
	"log"
	"time"

	_ "github.com/jackc/pgx/v5/stdlib"
)

type Book struct {
	ID    int64
	Title string
	Year  int
}

func main() {
	db, err := sql.Open("pgx", "postgres://shop:shop@localhost:5432/shop")
	if err != nil {
		log.Fatal(err)
	}
	defer db.Close()
	db.SetMaxOpenConns(20)
	db.SetConnMaxLifetime(30 * time.Minute)

	ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
	defer cancel()

	rows, err := db.QueryContext(ctx, "SELECT id, title, year FROM books WHERE year >= $1 ORDER BY year", 2010)
	if err != nil {
		log.Fatal(err)
	}
	defer rows.Close()
	var books []Book
	for rows.Next() {
		var b Book
		if err := rows.Scan(&b.ID, &b.Title, &b.Year); err != nil {
			log.Fatal(err)
		}
		books = append(books, b)
	}
	if err := rows.Err(); err != nil {
		log.Fatal(err)
	}
	log.Println(len(books))
}
```
! `*sql.DB` — пул соединений, один на приложение
! После `QueryContext` — `defer rows.Close()` и проверка `rows.Err()`
! Параметры только через `$1`, `$2` — никакой склейки строк

@lesson pgx, ошибки и NULL
Драйвер pgx можно использовать напрямую через `pgxpool` — он быстрее и знает типы PostgreSQL (массивы, JSONB, UUID). `pgx.CollectRows` собирает строки в срез структур.
Отсутствие строки — особая ошибка: `errors.Is(err, pgx.ErrNoRows)` (или `sql.ErrNoRows`) превращают в 404, а не в 500. Ошибку уникальности узнают по коду PostgreSQL `23505` и отдают 409.
`NULL` в столбце не влезет в обычный `string`: используют указатели `*string` или типы `sql.NullString`/`pgtype.Text`.
```code
package store

import (
	"context"
	"errors"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgconn"
	"github.com/jackc/pgx/v5/pgxpool"
)

var ErrNotFound = errors.New("not found")
var ErrDuplicate = errors.New("duplicate")

type User struct {
	ID    int64   `db:"id"`
	Email string  `db:"email"`
	Bio   *string `db:"bio"`        // может быть NULL
}

type Store struct{ pool *pgxpool.Pool }

func (s *Store) UserByID(ctx context.Context, id int64) (User, error) {
	rows, _ := s.pool.Query(ctx, "SELECT id, email, bio FROM users WHERE id = $1", id)
	u, err := pgx.CollectOneRow(rows, pgx.RowToStructByName[User])
	if errors.Is(err, pgx.ErrNoRows) {
		return User{}, ErrNotFound
	}
	return u, err
}

func (s *Store) CreateUser(ctx context.Context, email string) (int64, error) {
	var id int64
	err := s.pool.QueryRow(ctx, "INSERT INTO users (email) VALUES ($1) RETURNING id", email).Scan(&id)
	var pgErr *pgconn.PgError
	if errors.As(err, &pgErr) && pgErr.Code == "23505" {
		return 0, ErrDuplicate
	}
	return id, err
}
```
! «Нет строки» — не ошибка сервера, а 404
! Код `23505` — нарушение уникальности, отдавайте 409
! NULL-столбцы — в указатели или Null-типы

@lesson Транзакции и миграции
Транзакция: `tx, err := pool.Begin(ctx)`, затем запросы через `tx`, в конце `tx.Commit(ctx)`. Сразу после Begin пишут `defer tx.Rollback(ctx)` — после успешного Commit откат ничего не сделает, а при ошибке или панике отменит изменения.
Для операций со складом и деньгами блокируют строки `SELECT ... FOR UPDATE` или используют атомарный `UPDATE ... SET stock = stock - $1 WHERE id = $2 AND stock >= $1` и проверяют число изменённых строк.
Схему меняют миграциями (golang-migrate, goose): пары файлов `0001_init.up.sql` / `.down.sql` применяются по порядку и хранятся в git.
```code
func (s *Store) Reserve(ctx context.Context, productID int64, qty int) error {
	tx, err := s.pool.Begin(ctx)
	if err != nil {
		return err
	}
	defer tx.Rollback(ctx)            // безопасно и после Commit

	tag, err := tx.Exec(ctx,
		"UPDATE products SET stock = stock - $1 WHERE id = $2 AND stock >= $1", qty, productID)
	if err != nil {
		return err
	}
	if tag.RowsAffected() == 0 {
		return ErrOutOfStock           // откат через defer
	}
	if _, err := tx.Exec(ctx,
		"INSERT INTO reservations (product_id, qty) VALUES ($1, $2)", productID, qty); err != nil {
		return err
	}
	return tx.Commit(ctx)
}

// migrations/0001_init.up.sql
// CREATE TABLE products (id bigserial PRIMARY KEY, name text NOT NULL, stock int NOT NULL CHECK (stock >= 0));
// migrate -path migrations -database "$DATABASE_URL" up
```
! `defer tx.Rollback(ctx)` сразу после `Begin`
! Атомарный UPDATE с условием защищает от продажи товара, которого нет
! Миграции up и down — в репозитории
? mc Зачем `defer tx.Rollback(ctx)` сразу после Begin?

- Это ускоряет Commit
- Это обязательное требование Go

@lesson sqlc и тесты с базой
sqlc генерирует типобезопасный Go-код из SQL-файлов: пишете запрос с комментарием `-- name: GetBook :one`, запускаете `sqlc generate` — получаете функцию `q.GetBook(ctx, id)` с правильными типами. SQL остаётся SQL, а ошибки в нём ловятся при генерации.
Хранилище прячут за интерфейсом (`type BookStore interface { ... }`): обработчики HTTP тестируют с фальшивой реализацией в памяти, а сам SQL — интеграционными тестами на настоящем PostgreSQL через testcontainers-go.
Табличные тесты (`[]struct{ name string; in; want }` и `t.Run`) — идиоматичный стиль Go.
```code
package store

import "testing"

func Slugify(s string) string {
	out := []rune{}
	for _, r := range s {
		switch {
		case r >= 'a' && r <= 'z', r >= '0' && r <= '9':
			out = append(out, r)
		case r >= 'A' && r <= 'Z':
			out = append(out, r+32)
		case r == ' ' || r == '-':
			if len(out) > 0 && out[len(out)-1] != '-' {
				out = append(out, '-')
			}
		}
	}
	return string(out)
}

func TestSlugify(t *testing.T) {
	tests := []struct {
		name, in, want string
	}{
		{"простой", "Hello World", "hello-world"},
		{"двойные пробелы", "Go  is fun", "go-is-fun"},
		{"цифры", "Top 10 Books", "top-10-books"},
	}
	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			if got := Slugify(tt.in); got != tt.want {
				t.Errorf("Slugify(%q) = %q, want %q", tt.in, got, tt.want)
			}
		})
	}
}

// queries.sql для sqlc:
// -- name: GetBook :one
// SELECT id, title, year FROM books WHERE id = $1;
```
! Табличные тесты с `t.Run` — стандарт Go
! Обработчики тестируйте с фальшивым хранилищем, SQL — на настоящем PostgreSQL
! sqlc ловит ошибки в SQL ещё до запуска
